tsx
{/* Parse hoursDetail for today's day */}
{business.hoursDetail && (() => {
  try {
    const hours = JSON.parse(business.hoursDetail);
    const today = new Date().toLocaleDateString("en-US", { weekday: "short" });
    const todayHours = hours[today];
    if (todayHours) {
      const closeTime = todayHours.close ?? todayHours.split("-")[1]?.trim();
      return <p className="text-xs text-slate-500 mt-1">
        {business.openNow ? `Closes ${closeTime}` : `Opens ${todayHours.open ?? "—"}`}
      </p>;
    }
  } catch {}
  return null;
})()}
